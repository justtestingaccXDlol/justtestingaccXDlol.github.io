Expiry and accounts

curl

# register
curl -X POST https://your-service.onrender.com/data/binary/register \
  -H "Content-Type: application/json" -d '{"sid":"1234"}'

# write
curl -X POST https://your-service.onrender.com/data/binary \
  -H "SID: 1234" -H "Content-Type: application/json" -d '{"value":"10101010"}'

# read
curl https://your-service.onrender.com/data/binary -H "SID: 1234"

JavaScript

const r = await fetch("https://your-service.onrender.com/data/binary/3", {
  method: "POST",
  headers: { SID: "1234", "Content-Type": "application/json" },
  body: JSON.stringify({ value: "00001111" })
});
console.log(await r.json()); // { value: "00001111" }

Roblox (HttpService)

local HttpService = game:GetService("HttpService")
local res = HttpService:RequestAsync({
  Url = "https://your-service.onrender.com/data/binary",
  Method = "GET",
  Headers = { ["SID"] = "1234" },
})
print(HttpService:JSONDecode(res.Body).value)
